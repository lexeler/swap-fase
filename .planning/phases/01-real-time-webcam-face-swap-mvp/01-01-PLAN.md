def preload_cuda_libs() -> None
    """Call onnxruntime.preload_dlls(cuda=True, cudnn=True) before any session is created. No-op-safe to call once at startup."""

def select_providers(prefer_gpu: bool = True) -> list[str]
    """Returns ['CUDAExecutionProvider','CPUExecutionProvider'] if CUDA is available and prefer_gpu, else ['CPUExecutionProvider']."""

def active_provider(session) -> str
    """Returns session.get_providers()[0] — the provider that ACTUALLY bound, not just what was requested."""

def verify_gpu(raise_on_cpu: bool = False) -> str
    """Builds a tiny session, runs a warm-up inference, returns the active provider string ('CUDAExecutionProvider' or 'CPUExecutionProvider'). If raise_on_cpu and CPU, raise RuntimeError."""
